# --- OCR engine ------------------------------------------------------------
# paddlepaddle is the runtime; paddleocr is the model/pipeline layer.
#
# 3.x is required, not merely preferred. paddlepaddle 2.6.2's bundled arm64
# BLAS kernel faults inside cblas_sgemm during convolution on Apple Silicon,
# and Paddle's own crash handler then spins forever symbolising the stack -
# the process pins a core and never returns rather than raising. 3.x has
# working arm64 kernels. The provider still speaks both API dialects, so
# pinning back to 2.x on x86 remains possible if you need it.
#
# On first run the models are downloaded to ~/.paddlex/official_models
# (about 100 MB for English plus Arabic). See README "First run".
paddlepaddle>=3.0,<4.0
paddleocr>=3.0,<4.0
